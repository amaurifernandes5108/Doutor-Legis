## 🔑 IMPLEMENTAÇÃO NO SISTEMA

### 1. Via Backend API (Recomendado)

```python
# Adicionar ao server.py

@api_router.get("/prompt/mestre")
async def obter_prompt_mestre(current_user: User = Depends(get_current_user)):
    """Retorna prompt mestre personalizado para o usuário"""
    
    plano_config = get_plano_config(current_user.plan)
    
    # Verificar limites
    limite_consultas = verificar_limite_consultas(current_user.plan, current_user.consultas_mes_atual)
    limite_pdfs = verificar_limite_pdfs(current_user.plan, current_user.pdfs_mes_atual)
    
    # Ler prompt base
    with open('/app/PROMPT_MESTRE_DOUTOR_LEGIS.md', 'r', encoding='utf-8') as f:
        prompt_base = f.read()
    
    # Substituir variáveis
    prompt_personalizado = prompt_base.format(
        nome=current_user.name,
        email=current_user.email,
        plano=current_user.plan,
        plano_nome=plano_config.nome,
        limite_consultas=plano_config.consultas_mes or "Ilimitadas",
        consultas_mes_atual=current_user.consultas_mes_atual,
        limite_pdfs=plano_config.pdfs_mes or "Ilimitados",
        pdfs_mes_atual=current_user.pdfs_mes_atual,
        stripe_customer_id=current_user.stripe_customer_id or "N/A"
    )
    
    return {
        "prompt": prompt_personalizado,
        "usuario": {
            "nome": current_user.name,
            "email": current_user.email,
            "plano": plano_config.nome
        },
        "uso": {
            "consultas": limite_consultas,
            "pdfs": limite_pdfs
        },
        "features": plano_config.to_dict()
    }
```
